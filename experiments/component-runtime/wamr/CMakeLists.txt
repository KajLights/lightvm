cmake_minimum_required(VERSION 3.14)
project(component_wamr C)
if(NOT WAMR_SOURCE)
  message(FATAL_ERROR "Set WAMR_SOURCE to the pinned WAMR source directory")
endif()
string(TOLOWER "${CMAKE_SYSTEM_NAME}" WAMR_BUILD_PLATFORM)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)")
  set(WAMR_BUILD_TARGET AARCH64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)")
  set(WAMR_BUILD_TARGET X86_64)
else()
  message(FATAL_ERROR "Host prototype supports arm64 and x86_64 only")
endif()
set(WAMR_BUILD_INTERP 0)
set(WAMR_BUILD_AOT 1)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 0)
set(WAMR_BUILD_MEMORY_PROFILING 1)
set(WAMR_BUILD_ALLOC_WITH_USAGE 1)
set(WAMR_BUILD_SHRUNK_MEMORY 1)
set(WAMR_ROOT_DIR "${WAMR_SOURCE}")
include("${WAMR_SOURCE}/build-scripts/runtime_lib.cmake")
add_library(wamr_runtime STATIC ${WAMR_RUNTIME_LIB_SOURCE})
add_executable(wamr_probe host.c)
target_include_directories(wamr_probe PRIVATE "${WAMR_SOURCE}/core/iwasm/include")
target_compile_options(wamr_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(wamr_probe wamr_runtime m dl pthread)
