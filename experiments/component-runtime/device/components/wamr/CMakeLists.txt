# Isolated ESP-IDF AOT-only component; upstream source is pinned by build_wamr.py.
if(NOT CMAKE_BUILD_EARLY_EXPANSION)
  get_filename_component(CR_REPO "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
  if(NOT WAMR_SOURCE)
    set(WAMR_SOURCE "${CR_REPO}/build/component-runtime/deps/wamr-2.4.5")
  endif()
  if(NOT EXISTS "${WAMR_SOURCE}/build-scripts/runtime_lib.cmake")
    message(FATAL_ERROR "Pinned WAMR 2.4.5 source missing: ${WAMR_SOURCE}")
  endif()
  set(WAMR_BUILD_PLATFORM esp-idf)
  set(WAMR_BUILD_TARGET RISCV32_ILP32)
  set(WAMR_BUILD_INTERP 0)
  set(WAMR_BUILD_AOT 1)
  set(WAMR_BUILD_JIT 0)
  set(WAMR_BUILD_FAST_JIT 0)
  set(WAMR_BUILD_LIBC_BUILTIN 0)
  set(WAMR_BUILD_LIBC_WASI 0)
  set(WAMR_BUILD_LIB_PTHREAD 0)
  set(WAMR_BUILD_SHARED_MEMORY 0)
  set(WAMR_BUILD_SIMD 0)
  set(WAMR_BUILD_REF_TYPES 0)
  set(WAMR_BUILD_MULTI_MODULE 0)
  set(WAMR_BUILD_SHRUNK_MEMORY 1)
  set(WAMR_DISABLE_HW_BOUND_CHECK 1)
  set(WAMR_BUILD_MEMORY_PROFILING 0)
  set(WAMR_ROOT_DIR "${WAMR_SOURCE}")
  include("${WAMR_SOURCE}/build-scripts/runtime_lib.cmake")
  set(cr_wamr_sources ${WAMR_RUNTIME_LIB_SOURCE})
  set(cr_wamr_includes "${IWASM_DIR}/include" "${UTILS_SHARED_DIR}"
      "${PLATFORM_SHARED_DIR}" "${PLATFORM_SHARED_DIR}/../include" "${IWASM_COMMON_DIR}")
endif()
idf_component_register(SRCS ${cr_wamr_sources} INCLUDE_DIRS ${cr_wamr_includes}
                       REQUIRES pthread lwip esp_timer)
target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-format -Wno-error)
target_compile_definitions(${COMPONENT_LIB} PUBLIC BUILD_TARGET_RISCV32_ILP32=1 WASM_ENABLE_AOT=1 WASM_ENABLE_INTERP=0)
