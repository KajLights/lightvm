cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(bytecode_poc C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
set(POC_PIXELS 200 CACHE STRING "Configured pixel count")
set(POC_GPIO 16 CACHE STRING "Configured pixel GPIO")
set(POC_BRIGHTNESS 100 CACHE STRING "Configured brightness limit")
set(POC_ORDER GRB CACHE STRING "Configured wire order")
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(bytecode_poc main.c ../device/runtime.c ../vm.c "${root}/platforms/pico/platform.c")
target_include_directories(bytecode_poc PRIVATE .. ../device "${root}/src" "${root}/platforms/pico")
target_compile_options(bytecode_poc PRIVATE -O3 -Wall -Wextra -Werror -ffp-contract=off)
target_compile_definitions(bytecode_poc PRIVATE POC_PIXELS=${POC_PIXELS} POC_GPIO=${POC_GPIO}
  POC_BRIGHTNESS=${POC_BRIGHTNESS} POC_ORDER="${POC_ORDER}" PICO_STACK_SIZE=8192
  PICO_STDIO_USB_STDOUT_TIMEOUT_US=20000 LIGHTVM_BOARD_CONFIG="${root}/platforms/pico/boards/pico_w.h")
target_link_libraries(bytecode_poc pico_stdlib pico_stdio_usb pico_unique_id pico_rand hardware_pio hardware_flash hardware_sync hardware_watchdog)
pico_generate_pio_header(bytecode_poc "${root}/platforms/pico/ws2812.pio")
pico_set_double_implementation(bytecode_poc compiler)
pico_enable_stdio_usb(bytecode_poc 1)
pico_enable_stdio_uart(bytecode_poc 0)
target_link_options(bytecode_poc PRIVATE "LINKER:-T,${CMAKE_CURRENT_LIST_DIR}/storage.ld")
pico_add_extra_outputs(bytecode_poc)
