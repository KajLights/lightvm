cmake_minimum_required(VERSION 3.18)
option(LIGHTVM_WIFI "Include native Wi-Fi" OFF)
option(LIGHTVM_HTTP "Include HTTP management" OFF)
option(LIGHTVM_WEBSOCKET "Include plain WebSocket management (requires HTTP)" OFF)
if(LIGHTVM_WEBSOCKET AND NOT LIGHTVM_HTTP)
    message(FATAL_ERROR "WebSocket management requires LIGHTVM_HTTP=ON")
endif()
option(LIGHTVM_BOOT_SEQUENCE "Include persistent power-cycle sequence detection" OFF)
set(LIGHTVM_PLATFORM "host" CACHE STRING "Native port: host or pico")
set_property(CACHE LIGHTVM_PLATFORM PROPERTY STRINGS host pico)
if(LIGHTVM_PLATFORM STREQUAL "esp32")
    message(FATAL_ERROR "ESP32 uses its SDK project: idf.py -C platforms/esp32 -B <absolute-build-dir> -D LIGHTVM_BOARD=<board> build")
endif()
set(LIGHTVM_PLATFORM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/platforms/${LIGHTVM_PLATFORM}"
    CACHE PATH "Native port directory (may be outside the source tree)")
set(LIGHTVM_DEFAULT_BOARD_CONFIG "${LIGHTVM_PLATFORM_DIR}/board_config.h")
if(LIGHTVM_PLATFORM STREQUAL "pico" AND PICO_BOARD STREQUAL "pico_w")
    set(LIGHTVM_DEFAULT_BOARD_CONFIG "${LIGHTVM_PLATFORM_DIR}/boards/pico_w.h")
endif()
set(LIGHTVM_BOARD_CONFIG "${LIGHTVM_DEFAULT_BOARD_CONFIG}"
    CACHE FILEPATH "Board wiring and engine resource profile")
if(NOT EXISTS "${LIGHTVM_PLATFORM_DIR}/CMakeLists.txt" OR
   NOT EXISTS "${LIGHTVM_BOARD_CONFIG}")
    message(FATAL_ERROR "Port '${LIGHTVM_PLATFORM}' is not implemented or its board profile is missing")
endif()
# SDK/toolchain imports belong to the port and run before compiler detection.
if(EXISTS "${LIGHTVM_PLATFORM_DIR}/pre_project.cmake")
    include("${LIGHTVM_PLATFORM_DIR}/pre_project.cmake")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/lightvm_release.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/lightvm_storage.cmake")
project(lightvm C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/lightvm_core.cmake")
lightvm_add_core()
add_executable(lightvm)
target_compile_options(lightvm PRIVATE -Wall -Wextra -Werror)
target_link_libraries(lightvm PRIVATE lightvm_core)
# Include in the executable's directory: Pico SDK output helpers attach
# post-build commands and require that CMake directory scope.
include("${LIGHTVM_PLATFORM_DIR}/CMakeLists.txt")
if(LIGHTVM_PLATFORM STREQUAL "host")
    include(CTest)
    if(BUILD_TESTING)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        foreach(fixture bytecode portable)
            add_test(NAME ${fixture}_fixtures_current COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_SOURCE_DIR}/tests/generate_${fixture}_fixtures.py" --check)
        endforeach()
        add_executable(test_release tests/test_release.c)
        target_compile_options(test_release PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(test_release PRIVATE lightvm_core)
        add_test(NAME release_compatibility COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_SOURCE_DIR}/tests/run_release_contract.py" $<TARGET_FILE:test_release>)
        add_executable(test_platform tests/test_platform.c)
        target_compile_options(test_platform PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(test_platform PRIVATE lightvm_core)
        add_test(NAME platform_contract COMMAND test_platform)
        add_executable(test_storage tests/test_storage.c platforms/host/platform.c)
        target_compile_options(test_storage PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(test_storage PRIVATE lightvm_core)
        add_test(NAME storage_backends COMMAND test_storage)
        add_executable(test_composite tests/test_composite.c)
        target_compile_options(test_composite PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(test_composite PRIVATE lightvm_core)
        add_test(NAME compositing_precision COMMAND test_composite)
        if(LIGHTVM_BOOT_SEQUENCE)
            add_executable(test_boot_sequence tests/test_boot_sequence.c)
            target_compile_options(test_boot_sequence PRIVATE -Wall -Wextra -Werror)
            target_link_libraries(test_boot_sequence PRIVATE lightvm_core)
            add_test(NAME boot_sequence_journal COMMAND test_boot_sequence)
        endif()
        add_executable(test_network tests/test_network.c)
        target_compile_options(test_network PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(test_network PRIVATE lightvm_core)
        add_test(NAME network_policy COMMAND test_network)
        add_executable(test_dhcp tests/test_dhcp.c src/dhcp.c)
        target_include_directories(test_dhcp PRIVATE src)
        target_compile_options(test_dhcp PRIVATE -Wall -Wextra -Werror)
        add_test(NAME ap_dhcp COMMAND test_dhcp)
        if(LIGHTVM_HTTP)
            add_executable(test_discovery tests/test_discovery.c)
            target_link_libraries(test_discovery PRIVATE lightvm_core)
            target_compile_options(test_discovery PRIVATE -Wall -Wextra -Werror)
            add_test(NAME mdns_discovery COMMAND test_discovery)
            add_executable(test_http_core tests/test_http_core.c platforms/host/platform.c)
            target_link_libraries(test_http_core PRIVATE lightvm_core)
            target_compile_options(test_http_core PRIVATE -Wall -Wextra -Werror)
            add_test(NAME http_contract COMMAND test_http_core)
        endif()
        if(LIGHTVM_WEBSOCKET)
            add_executable(test_websocket_core tests/test_websocket_core.c platforms/host/platform.c)
            target_link_libraries(test_websocket_core PRIVATE lightvm_core)
            target_compile_options(test_websocket_core PRIVATE -Wall -Wextra -Werror)
            add_test(NAME websocket_contract COMMAND test_websocket_core)
        endif()
    endif()
endif()

# Keep redistribution notices beside the generated executables/firmware.
configure_file(LICENSE "${CMAKE_CURRENT_BINARY_DIR}/LICENSE" COPYONLY)
configure_file(THIRD_PARTY_NOTICES.md "${CMAKE_CURRENT_BINARY_DIR}/THIRD_PARTY_NOTICES.md" COPYONLY)
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/licenses" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
