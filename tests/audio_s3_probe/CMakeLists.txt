cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET "esp32s3" CACHE STRING "Target")
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "Local configuration")
if(NOT KAJ_AUDIO_CORE OR NOT EXISTS "${KAJ_AUDIO_CORE}/features.c")
    message(FATAL_ERROR "Set KAJ_AUDIO_CORE to the pinned kaj-fi core directory")
endif()
get_filename_component(KAJ_FI_ROOT "${KAJ_AUDIO_CORE}" DIRECTORY)
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/../../kaj-audio.rev" expected_revision LIMIT_COUNT 1)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../../tools/audio_dependency.py" --verify "${KAJ_AUDIO_CORE}"
    OUTPUT_VARIABLE actual_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual_revision STREQUAL expected_revision)
    message(FATAL_ERROR "S3 probe requires the exact kaj-audio.rev checkout")
endif()
include("$ENV{IDF_PATH}/tools/cmake/version.cmake")
if(NOT "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" VERSION_EQUAL "5.5.3")
    message(FATAL_ERROR "S3 probe requires ESP-IDF 5.5.3")
endif()
set(EXTRA_COMPONENT_DIRS "${KAJ_FI_ROOT}/components/esp-dsp")
set(ENV{IDF_COMPONENT_MANAGER} "0")
set(COMPONENTS main)
include("$ENV{IDF_PATH}/tools/cmake/project.cmake")
project(audio_s3_probe)
