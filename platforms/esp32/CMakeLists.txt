cmake_minimum_required(VERSION 3.18)

option(LIGHTVM_WIFI "Include native Wi-Fi" OFF)
option(LIGHTVM_HTTP "Include HTTP management" OFF)
option(LIGHTVM_BOOT_SEQUENCE "Include persistent power-cycle sequence detection" OFF)
if(LIGHTVM_HTTP AND NOT LIGHTVM_WIFI)
    message(FATAL_ERROR "Embedded HTTP requires LIGHTVM_WIFI=ON")
endif()
# IDF discovers component requirements in a separate early-expansion process.
# Propagate the build switches to that process before it resolves dependencies.
if(LIGHTVM_WIFI)
    set(ENV{LIGHTVM_PORT_WIFI} "1")
else()
    set(ENV{LIGHTVM_PORT_WIFI} "0")
endif()
if(LIGHTVM_HTTP)
    set(ENV{LIGHTVM_PORT_HTTP} "1")
else()
    set(ENV{LIGHTVM_PORT_HTTP} "0")
endif()
get_filename_component(LIGHTVM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LIGHTVM_BOARD "" CACHE STRING "XIAO board: xiao_esp32c3, xiao_esp32s3 or xiao_esp32c6")
set_property(CACHE LIGHTVM_BOARD PROPERTY STRINGS xiao_esp32c3 xiao_esp32s3 xiao_esp32c6)
if(LIGHTVM_BOARD STREQUAL "xiao_esp32c3")
    set(LIGHTVM_IDF_TARGET esp32c3)
elseif(LIGHTVM_BOARD STREQUAL "xiao_esp32s3")
    set(LIGHTVM_IDF_TARGET esp32s3)
elseif(LIGHTVM_BOARD STREQUAL "xiao_esp32c6")
    set(LIGHTVM_IDF_TARGET esp32c6)
else()
    message(FATAL_ERROR "Set LIGHTVM_BOARD=xiao_esp32c3, xiao_esp32s3 or xiao_esp32c6")
endif()
if((DEFINED IDF_TARGET AND NOT IDF_TARGET STREQUAL LIGHTVM_IDF_TARGET) OR
   (DEFINED ENV{IDF_TARGET} AND NOT "$ENV{IDF_TARGET}" STREQUAL LIGHTVM_IDF_TARGET))
    message(FATAL_ERROR "LIGHTVM_BOARD=${LIGHTVM_BOARD} requires IDF_TARGET=${LIGHTVM_IDF_TARGET}")
endif()
if(DEFINED LIGHTVM_CONFIGURED_BOARD AND NOT LIGHTVM_CONFIGURED_BOARD STREQUAL LIGHTVM_BOARD)
    message(FATAL_ERROR "Use a separate build directory for each LIGHTVM_BOARD")
endif()
set(LIGHTVM_CONFIGURED_BOARD "${LIGHTVM_BOARD}" CACHE INTERNAL "Board in this build directory")
set(IDF_TARGET "${LIGHTVM_IDF_TARGET}" CACHE STRING "ESP-IDF target")
set(LIGHTVM_BOARD_CONFIG "${CMAKE_CURRENT_LIST_DIR}/boards/${LIGHTVM_BOARD}.h"
    CACHE FILEPATH "Board wiring and engine resource profile")
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "Build-local ESP-IDF configuration")
# ESP-IDF automatically loads sdkconfig.defaults.<target> alongside this file.
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
include("${LIGHTVM_ROOT}/cmake/lightvm_storage.cmake")
# A custom table may reserve OTA or other use-case-specific partitions. Without
# one, generate the existing layout with the requested storage capacity.
set(LIGHTVM_PARTITION_TABLE "" CACHE FILEPATH "Custom ESP32 partition table (optional)")
if(LIGHTVM_PARTITION_TABLE)
    get_filename_component(LIGHTVM_PARTITION_FILENAME "${LIGHTVM_PARTITION_TABLE}" ABSOLUTE)
    if(NOT EXISTS "${LIGHTVM_PARTITION_FILENAME}")
        message(FATAL_ERROR "LIGHTVM_PARTITION_TABLE does not exist")
    endif()
else()
    set(LIGHTVM_PARTITION_FILENAME "${CMAKE_BINARY_DIR}/lightvm-partitions.csv")
    set(LIGHTVM_PARTITION_TEXT "# Name, Type, SubType, Offset, Size, Flags\nnvs, data, nvs, 0x9000, 0x6000,\nphy_init, data, phy, 0xf000, 0x1000,\nfactory, app, factory, 0x10000, 1M,\nlightvm, data, 0x40, 0x110000, ${LIGHTVM_STORAGE_BYTES_VALUE},\n")
    if(LIGHTVM_BOOT_SEQUENCE)
        math(EXPR LIGHTVM_BOOT_OFFSET "0x110000 + ${LIGHTVM_STORAGE_BYTES_VALUE}" OUTPUT_FORMAT HEXADECIMAL)
        string(APPEND LIGHTVM_PARTITION_TEXT "lightvm_boot, data, 0x41, ${LIGHTVM_BOOT_OFFSET}, 8K,\n")
    endif()
    file(WRITE "${LIGHTVM_PARTITION_FILENAME}" "${LIGHTVM_PARTITION_TEXT}")
endif()
set(LIGHTVM_PARTITION_DEFAULTS "${CMAKE_BINARY_DIR}/sdkconfig.boot_sequence.defaults")
file(WRITE "${LIGHTVM_PARTITION_DEFAULTS}" "CONFIG_PARTITION_TABLE_CUSTOM_FILENAME=\"${LIGHTVM_PARTITION_FILENAME}\"\n")
list(APPEND SDKCONFIG_DEFAULTS "${LIGHTVM_PARTITION_DEFAULTS}")
if(EXISTS "${SDKCONFIG}")
    file(READ "${SDKCONFIG}" LIGHTVM_SDKCONFIG_TEXT)
    string(REGEX REPLACE "CONFIG_PARTITION_TABLE_CUSTOM_FILENAME=\"[^\"]*\""
        "CONFIG_PARTITION_TABLE_CUSTOM_FILENAME=\"${LIGHTVM_PARTITION_FILENAME}\"" LIGHTVM_UPDATED_SDKCONFIG "${LIGHTVM_SDKCONFIG_TEXT}")
    if(NOT LIGHTVM_UPDATED_SDKCONFIG STREQUAL LIGHTVM_SDKCONFIG_TEXT)
        file(WRITE "${SDKCONFIG}" "${LIGHTVM_UPDATED_SDKCONFIG}")
    endif()
endif()
if(NOT EXISTS "$ENV{IDF_PATH}/tools/cmake/project.cmake")
    message(FATAL_ERROR "Activate ESP-IDF v5.5.3 with its export.sh first")
endif()
include("$ENV{IDF_PATH}/tools/cmake/version.cmake")
set(IDF_VERSION "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}")
if(NOT IDF_VERSION VERSION_EQUAL "5.5.3")
    message(FATAL_ERROR "This port is pinned to ESP-IDF 5.5.3; found ${IDF_VERSION}")
endif()
# All dependencies come from the pinned SDK. Avoid
# managed-component lockfiles in the source tree and build only this component's
# dependency closure rather than unused networking examples/services.
set(ENV{IDF_COMPONENT_MANAGER} "0")
include("${LIGHTVM_ROOT}/cmake/lightvm_release.cmake")
set(PROJECT_VER "${LIGHTVM_VERSION}")
set(COMPONENTS main)
include("$ENV{IDF_PATH}/tools/cmake/project.cmake")
project(lightvm)

configure_file("${LIGHTVM_ROOT}/LICENSE" "${CMAKE_BINARY_DIR}/LICENSE" COPYONLY)
configure_file("${LIGHTVM_ROOT}/THIRD_PARTY_NOTICES.md" "${CMAKE_BINARY_DIR}/THIRD_PARTY_NOTICES.md" COPYONLY)
file(COPY "${LIGHTVM_ROOT}/licenses" DESTINATION "${CMAKE_BINARY_DIR}")
