set(port_sources main.c platform.c ws2812_encoder.c)
if(LIGHTVM_MQTT)
    list(APPEND port_sources "${LIGHTVM_ROOT}/src/mqtt_posix.c")
endif()
set(port_requires esp_driver_gpio esp_driver_rmt esp_driver_usb_serial_jtag esp_timer esp_partition)
if(LIGHTVM_MQTT)
    list(APPEND port_requires pthread)
endif()
if("$ENV{LIGHTVM_PORT_WIFI}" STREQUAL "1")
    list(APPEND port_sources "${LIGHTVM_ROOT}/shared/device-core/ports/esp32/network_port.c")
    list(APPEND port_requires esp_wifi esp_netif esp_event nvs_flash esp_hw_support lwip)
endif()
if("$ENV{LIGHTVM_PORT_HTTP}" STREQUAL "1")
    list(APPEND port_requires esp_http_server)
endif()
idf_component_register(SRCS ${port_sources} INCLUDE_DIRS "." REQUIRES ${port_requires})
include("${LIGHTVM_ROOT}/cmake/lightvm_core.cmake")
lightvm_add_core()
if(LIGHTVM_ESP32_SELECTED_PROFILE STREQUAL "c3")
    target_compile_definitions(lightvm_config INTERFACE LIGHTVM_ESP32_C3_PROFILE=1)
    # Optimise the interpreter/kernels without changing IDF or floating-point
    # semantics. Keep the core's -ffp-contract=off and the SDK's ISR options.
    target_compile_options(lightvm_core PRIVATE -O3)
endif()
target_link_libraries(${COMPONENT_LIB} PUBLIC lightvm_core)
target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror)
# Ordinary libraries must inherit the IDF compile options too (notably Xtensa
# long calls and section placement). No SDK dependency enters the shared source.
target_link_libraries(lightvm_core PRIVATE idf::esp_common)
target_link_libraries(device_core PRIVATE idf::esp_common)
